awk -F'\t' '{if ($0 ~ /^#/) print $0; else if ($4 != "." && $4 != "N" && $5 !~ /(^|\.|,)N(,|$)/ && $5 !~ /(^|\.|,)\.(,|$)/) {remove=0; for (i=10; i<=NF; i++) if ($i ~ /\./) {remove=1; break} if (!remove) print}}' Falk-k15-weeded-m90-mapped-to-OM181945.1.map.vcf > Falk-k15-weeded-m90-mapped-to-OM181945.1-clean.map.vcf
